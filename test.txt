var maxDepth = function(root) {
    if(root == undefined) return 0;
    let level = 1;
    level = checkIfIsLeaf(root.val,root.left,root.right,level );    
    return level;
};
function checkIfIsLeaf(val, left, right, level)
{
    if(val == null) return;
    console.log(val,left,right,level)    
    level++;
    if(left == null && right == null)
        return level;
    
    if(left != null)
    {
        if(left.left != null || left.right != null)
        {
            return checkIfIsLeaf(left.val, left.left, left.right, level)
        }
    }
    if(right != null)
    {
        if(right.left != null || right.right != null)
        {
            return checkIfIsLeaf(right.val, right.left, right.right, level)
        }       
    }
    return level;
}
